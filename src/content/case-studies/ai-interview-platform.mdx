class InterviewProctor {
  constructor(sessionId, websocket) {
    this.sessionId = sessionId;
    this.ws = websocket;
    this.mediaStream = null;
    this.isMonitoring = false;
  }

  async startProctoring() {
    try {
      // Request camera and microphone access
      this.mediaStream = await navigator.mediaDevices.getUserMedia({
        video: { width: 1280, height: 720 },
        audio: true
      });

      // Initialize face detection
      await this.initializeFaceDetection();
      
      // Start monitoring
      this.isMonitoring = true;
      this.monitor();
      
    } catch (error) {
      console.error('Proctoring initialization failed:', error);
      this.handleProctoringError(error);
    }
  }

  monitor() {
    if (!this.isMonitoring) return;

    // Check for multiple faces
    this.detectFaces().then(faces => {
      if (faces.length > 1) {
        this.reportViolation('multiple_faces_detected', {
          count: faces.length,
          timestamp: Date.now()
        });
      } else if (faces.length === 0) {
        this.reportViolation('face_not_visible', {
          timestamp: Date.now()
        });
      }
    });

    // Monitor tab switching
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        this.reportViolation('tab_switch', {
          timestamp: Date.now()
        });
      }
    });

    // Continue monitoring
    setTimeout(() => this.monitor(), 2000);
  }

  reportViolation(type, data) {
    this.ws.send(JSON.stringify({
      type: 'proctoring_event',
      event: { type, data, sessionId: this.sessionId }
    }));
  }
}

class FairEvaluationEngine:
    def __init__(self):
        self.models = [
            TechnicalAccuracyModel(),
            ProblemSolvingModel(), 
            CommunicationModel()
        ]
        self.bias_detector = BiasDetectionSystem()
        
    async def evaluate_with_fairness(self, response_data):
        # Multi-model evaluation
        evaluations = await asyncio.gather(*[
            model.evaluate(response_data) for model in self.models
        ])
        
        # Bias detection and correction
        bias_report = self.bias_detector.analyze(evaluations, response_data)
        if bias_report.has_bias:
            evaluations = self.correct_bias(evaluations, bias_report)
            
        # Weighted final score
        return self.compute_final_score(evaluations)
